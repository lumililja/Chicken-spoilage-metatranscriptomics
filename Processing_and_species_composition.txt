##4-degree data

## Removing 16S and 18S rRNA from the samples

#!/bin/bash
#

for f in *.fastq.gz
do

ext=${f##*.}
rootname=${f%.*}
basename=${rootname%.*}

export PATH=$PATH:/users/julimann/Applications/SortMeRNA/bin

sortmerna --workdir /users/julimann/Scratch/Metagenomics/SortMeRNA/$basename --ref /users/julimann/Applications/SortMeRNA/data/smr_v4.3_sensitive_db.fasta --reads "$f" --aligned $basename.rRNA-hits --other  $basename.CDS-hit -a 1  --fastx

done

## Removing chicken/broiler reads from the data

## building the index for bowtie2

bowtie2-build GCA_016699485.1_bGalGal1.mat.broiler.GRCg7b_genomic.fna chicken.btindex

## running bowtie2 with the index

#!/bin/bash
#

for f in *.CDS-hits.fq
do

ext=${f##*.}
rootname=${f%.*}
basename=${rootname%.*}
versionless=${basename%-*}

bowtie2 -x chicken.btindex --very-sensitive -U -q $f --un $basename.host_removed >> mapped_and_unmapped.sam

done

#!/bin/bash
#

for f in   A0{05..19}*.fastq.gz
do

ext=${f##*.}
rootname=${f%.*}
basename=${rootname%.*}

export PATH=$PATH:/users/julimann/Applications/SortMeRNA/bin

sortmerna --workdir /users/julimann/Scratch/Transcriptomics/6Degrees/SMR/$basename --ref /users/julimann/Applications/SortMeRNA/data/smr_v4.3_sensitive_db.fasta --reads $name"_R1_001.fastq.gz" --reads $name"_R2_001.fastq.gz" --paired_out
 --other $basename.fq.gz --out2 output2.$name.fq --fastx

done

## removing reads aligning to chicken
## the same chicken index from 4-degree sample was used

#!/bin/bash
#

for f in *fwd.fq.gz
do

ext=${f##*.}
rootname=${f%.*}
basename=${rootname%.*}
name=$(echo -n $basename | awk -F '[_]' '{print $1"_"$2"_"$3"_"$4"_"$5}')

bowtie2 --very-sensitive -q -x chicken.btindex -1 $name"_R1_fwd".fq.gz -2 $name"_R2_rev".fq.gz --un-conc $name.host_removed >> mapped_and_unmapped.sam

done
